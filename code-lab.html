C++

Orbital Physics 
#include <iostream>
#include <cmath>

int main() {

    double G = 6.67430e-11;
    double M = 5.972e24;
    double r = 6.771e6;

    double velocity =
        sqrt(G * M / r);

    std::cout
        << "Orbital velocity: "
        << velocity
        << " m/s";

    return 0;
}

Node.js

Space Data Server 
const http = require("http");

const server = http.createServer(
    (req, res) => {

        res.writeHead(200, {
            "Content-Type":
            "application/json"
        });

        res.end(JSON.stringify({
            mission: "ASTRONIUS-1",
            status: "ACTIVE",
            destination: "Mars"
        }));
    }
);

server.listen(3000);
